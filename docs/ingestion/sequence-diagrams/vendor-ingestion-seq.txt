loop for column in df.columns
  note right of Script
    if column.required:
      is_required_mask = df[column.name].isna()
      if is_require_mask.any():
        df.loc[is_require_mask, column.name+"_error"] = "required field"
        hasError = True
    if column.pattern:
      pattern_mask = ~df[column.name].str.fullMatch(rf"{column.pattern}")
      if pattern_mask.any():
        df.loc[pattern_mask, column.name+"_error"] = f"doesn't match the pattern: {column.pattern}"
        hasError = True
    if "mdmsSource" in column:
      if column["mdmsSource"].filterType == FilterType.ONE_OF:
        value_mask = ~df[column.name].isin(columnDataSourceMap[column.name])
        if value_mask.any():
          df.loc[value_mask, column.name+"_error"] = f"doesn't match the values: {columnDataSourceMap[column.name]}"
          hasError = True
  end note
end
